บทที่ 139 ทรัฟเฟิลยักษ์ใหญ่ (ขอโหวตรายเดือนและขอให้สมัครสมาชิก)

เซี่ยรั่วเฟยใช้พลั่วสนามขุดดินรอบทรัฟเฟิลออกอย่างระมัดระวัง ก่อนจะใช้มือเปล่าขุดทรัฟเฟิลขาวก้อนนี้ขึ้นมา

หลังปัดดินบนผิวทรัฟเฟิลออกคร่าวๆ เซี่ยรั่วเฟยก็ลองชั่งน้ำหนักด้วยมือ ก้อนนี้น่าจะหนักราว 400-500 กรัม และมีขนาดใหญ่พอๆ กับส้มโอหนึ่งลูก

ประเดิมได้งดงามจริงๆ!

เซี่ยรั่วเฟยดีใจเป็นอย่างยิ่ง

เขารู้ว่าทรัฟเฟิลยิ่งมีน้ำหนักต่อก้อนมากเท่าไรก็ยิ่งล้ำค่ามากเท่านั้น ก่อนหน้านี้เขาเคยค้นข้อมูลมาแล้วว่า ในปี 15 ทรัฟเฟิลขาวก้อนหนึ่งซึ่งหนัก 900 กรัม ถูกประมูลไปในราคา 100,000 ยูโร คิดเป็นเงินราว 680,000 สกุลเงินหัวเซี่ย

ทรัฟเฟิลขาวก้อนนี้หนักประมาณ 500 กรัม หากอ้างอิงราคาทรัฟเฟิลขาวที่มีน้ำหนักใกล้เคียงกันจากการประมูลครั้งนั้น มันก็น่าจะมีมูลค่าราว 200,000 สกุลเงินหัวเซี่ย

และนั่นยังเป็นกรณีที่คุณภาพทัดเทียมกันเท่านั้น

ให้ตายเซี่ยรั่วเฟยก็ไม่เชื่อว่าทรัฟเฟิลขาวที่ผลิตจากมิติภาพวิญญาณของตน จะมีคุณภาพด้อยกว่าทรัฟเฟิลขาวที่เติบโตตามธรรมชาติในโลกภายนอก—แค่ไม่ทิ้งห่างพวกมันหลายช่วงตัวก็นับว่าไว้หน้ามากแล้ว

ทรัฟเฟิลขาวก้อนนี้ช่วยเพิ่มความมั่นใจให้เซี่ยรั่วเฟยอย่างมาก

เขาครุ่นคิดเล็กน้อย ก่อนพาสายฟ้าออกจากมิติภาพวิญญาณ ให้สายฟ้ารออยู่ในห้อง ส่วนตัวเขาลงไปยังห้องเก็บของเล็กชั้นล่าง เมื่อล็อกประตูเรียบร้อยแล้วก็ส่งข้าวสารหลายกระสอบเข้าไปในมิติภาพวิญญาณทันที

หลังกลับมายังห้องนอนชั้นบน เซี่ยรั่วเฟยก็พาสายฟ้าเข้าไปในมิติภาพวิญญาณอีกครั้ง

เขาฉีกกระสอบข้าวสาร เทข้าวสารเป็นชั้นลงในกล่องเก็บของใบใหญ่ 2 ใบ จากนั้นใช้แปรงขนาดเล็กปัดดินบนทรัฟเฟิลทั้ง 2 ก้อนอย่างละเอียด ก่อนนำพวกมันใส่ลงในกล่องเก็บของแยกกัน แล้วฝังไว้ในข้าวสาร

สาเหตุที่เซี่ยรั่วเฟยทำเช่นนี้ เพราะทรัฟเฟิลส่งกลิ่นหอมเข้มข้นออกมาอยู่ตลอด แม้เขาจะรู้ว่าสิ่งของภายในมิติภาพวิญญาณจะไม่เสื่อมสภาพ แต่ไม่รู้ว่ากลิ่นหอมจะจางหายไปหรือไม่ ดังนั้นเก็บไว้ในข้าวสารและปิดผนึกภายในกล่องเก็บของจึงปลอดภัยกว่า

เมื่อทำทุกอย่างเสร็จ เซี่ยรั่วเฟยก็พาสายฟ้าออกตามหาตำแหน่งของทรัฟเฟิลแห่งถัดไป

แม้สายฟ้าจะไม่เคยผ่านการฝึกฝน แต่ประสาทรับกลิ่นของมันเฉียบคมกว่าสุนัขล่าทรัฟเฟิลเหล่านั้นมาก อีกทั้งยังรู้ความและเข้าใจคำสั่งของเซี่ยรั่วเฟย ดังนั้นประสิทธิภาพในการค้นหาทรัฟเฟิลของมันจึงสูงมาก

กลับเป็นประสิทธิภาพของเซี่ยรั่วเฟยที่ทำให้ความคืบหน้าล่าช้าลง

เพราะทุกครั้งที่ขุด เขาต้องระมัดระวังอย่างยิ่ง และหลังขุดทรัฟเฟิลขึ้นมาแล้ว ยังต้องปัดดินบนผิวอย่างระมัดระวังก่อนนำใส่กล่องเก็บของ

ดังนั้นทุกครั้งที่สายฟ้าหาตำแหน่งพบ มันก็จะหมอบพักอยู่ข้างๆ ส่วนเซี่ยรั่วเฟยต้องวุ่นอยู่ตลอดเวลา

ทรัฟเฟิลก้อนแล้วก้อนเล่าถูกเซี่ยรั่วเฟยขุดขึ้นมาจากใต้ดิน มีทั้งทรัฟเฟิลดำและทรัฟเฟิลขาวที่ล้ำค่ายิ่งกว่า

หลังผ่านการทำความสะอาด พวกมันก็ถูกนำใส่ลงในกล่องเก็บของอย่างต่อเนื่อง

ไม่นานกล่องเก็บของใบใหญ่ทั้ง 2 ใบก็เต็ม เซี่ยรั่วเฟยจึงหากล่องเก็บของมาเพิ่มอีก 2 ใบ แล้วเทข้าวสารลงไป

เขายุ่งอยู่เช่นนี้นานสิบกว่าชั่วโมง ในที่สุดเซี่ยรั่วเฟยก็ขุดทรัฟเฟิลดำก้อนสุดท้ายขึ้นมา หลังถามสายฟ้าและยืนยันว่าภายในป่าโอ๊กแห่งนี้ไม่มีทรัฟเฟิลเหลืออยู่อีกแม้แต่ก้อนเดียว เขาจึงถอนหายใจอย่างโล่งอก—งานนี้ไม่เพียงทดสอบกำลังร่างกาย แต่ยังทดสอบความอดทนของคนด้วย

แม้เซี่ยรั่วเฟยจะรู้สึกเหนื่อยล้าอย่างมาก อีกทั้งเนื้อตัวยังเปื้อนดินไปหมด แต่เมื่อมองทรัฟเฟิลเต็ม 4 กล่อง ในใจก็เปี่ยมด้วยความรู้สึกสำเร็จ

จำนวนของทรัฟเฟิลดำและทรัฟเฟิลขาวแทบจะเท่ากัน

กล่องเก็บของใบใหญ่ 2 ใบเต็มจนแน่น ส่วนอีก 2 ใบก็ใส่ไปเกินครึ่งแล้ว ส่วนจะเก็บเกี่ยวได้เท่าไรกันแน่ ยังต้องใช้เครื่องชั่งตั้งพื้นชั่งดูก่อนจึงจะรู้

แต่เซี่ยรั่วเฟยสามารถประเมินคร่าวๆ ได้

ในบรรดาทรัฟเฟิลเหล่านี้ ก้อนที่ใหญ่พอๆ กับทรัฟเฟิลขาวก้อนแรกที่สายฟ้าหาพบมีมากถึง 7-8 ก้อน หากคิดตามราคาประมูลของปีที่แล้ว แค่ทรัฟเฟิลตรงนี้ก็มีมูลค่าอย่างน้อยราว 2 ล้านแล้ว

นอกจากนี้ยังมีทรัฟเฟิลขนาดยักษ์อีก 3 ก้อน โดยก้อนที่ใหญ่ที่สุดมีขนาดใหญ่กว่าลูกบาสเกตบอลเล็กน้อย เมื่อลองชั่งด้วยมือก็น่าจะหนักเกือบ 4 จิน ส่วนอีก 2 ก้อนก็หนักกว่า 2 จินและ 3 จิน

หากเซี่ยรั่วเฟยจำไม่ผิด ก้อนที่ใหญ่ที่สุดอาจมีขนาดเหนือกว่าทรัฟเฟิลขาวที่ใหญ่ที่สุดในโลกเมื่อปี 14 เสียอีก ตอนนั้นทรัฟเฟิลขาวก้อนนั้นหนัก 16 ปอนด์ หรือประมาณ 89 กิโลกรัม

ทรัฟเฟิลก้อนที่เซี่ยรั่วเฟยขุดพบนี้ เรียกได้ว่าเป็นยักษ์ใหญ่โดยแท้จริง

มูลค่าของทรัฟเฟิลเช่นนี้ไม่อาจวัดด้วยราคาตลาดได้อีกต่อไป โดยเฉพาะหากนำเข้าร่วมงานประมูล ก็มีความเป็นไปได้อย่างยิ่งที่จะประมูลได้ราคาสูงลิ่ว!

สถานการณ์ของทรัฟเฟิลดำก็ใกล้เคียงกัน เพียงแต่โดยทั่วไปทรัฟเฟิลดำมีขนาดเล็กกว่าทรัฟเฟิลขาวเล็กน้อย ในบรรดาทรัฟเฟิลดำที่เซี่ยรั่วเฟยพบ ก้อนที่ใหญ่ที่สุดหนักเกือบ 3 จิน ซึ่งหาได้ยากอย่างยิ่งเช่นกัน

เซี่ยรั่วเฟยยืนครุ่นคิดอยู่ในมิติภาพวิญญาณครู่หนึ่ง รู้สึกว่าครั้งนี้มีทรัฟเฟิลที่ผลิตออกมามากเกินไป ไม่เหมาะจะปล่อยออกไปขายในคราวเดียว มิฉะนั้นราคาประมูลย่อมขึ้นไปไม่สูงแน่

เขาถึงขั้นคิดจะย้ายต้นโอ๊กเหล่านี้ไปปลูกในมิติภาพวิญญาณใหม่

สิ่งของที่ผลิตได้เฉพาะบางฤดูกาลของปี และไม่มีวิธีใดเก็บรักษาไว้ได้นานเช่นนี้ ต่อให้ผลิตออกมามากเพียงใดภายในมิติภาพวิญญาณก็ไม่มีความหมาย

แม้ในมิติภาพวิญญาณเดิมจะสามารถรักษาความสดได้อย่างสมบูรณ์ แต่ทรัฟเฟิลนั้นเป็นของที่ยิ่งหายากยิ่งมีค่าอยู่แล้ว หากปล่อยออกสู่ตลาดเป็นจำนวนมาก ย่อมก่อให้เกิดความผันผวนในตลาดและทำให้ราคาดิ่งลงในท้ายที่สุด เช่นนั้นย่อมได้ไม่คุ้มเสียอย่างแน่นอน

อีกทั้งหากจู่ๆ นำทรัฟเฟิลซึ่งไม่อาจเก็บรักษาไว้ได้นานออกขายในฤดูกาลที่ไม่มีผลผลิต เรื่องนี้ก็น่าสงสัยอย่างยิ่ง และอาจทำให้ความลับของตนถูกเปิดเผยได้โดยง่าย

แน่นอนว่าเซี่ยรั่วเฟยยังไม่รีบตัดสินใจว่าจะย้ายไปปลูกหรือไม่ การเก็บเกี่ยวครั้งนี้บังเอิญทันช่วงปลายฤดูกาลของทรัฟเฟิลพอดี ทรัฟเฟิลชุดนี้จึงยังสามารถนำออกขายได้ตามปกติ

ดังนั้นเรื่องเร่งด่วนที่สุดในตอนนี้คือวางแผนการขายทรัฟเฟิลชุดนี้ให้ดีเสียก่อน

อีก 2 วันเมื่อไปเซินเฉิง ทรัฟเฟิลชั้นยอดก้อนใหญ่ที่สุดหลายก้อนย่อมต้องนำไปประมูล สำหรับทรัฟเฟิลระดับนี้ สถานที่ที่เหมาะสมกับพวกมันที่สุดก็คืองานประมูล

ส่วนก้อนที่มีขนาดปานกลางซึ่งเหลืออยู่ ก็สามารถนำไปเป็นตัวประกอบด้วยเล็กน้อย

จากนั้นก้อนอื่นๆ ก็เก็บไว้ก่อน รอจนกลับถึงซานซานแล้วค่อยขายผ่านช่องทางอื่น—ทรัฟเฟิลคุณภาพดีถึงเพียงนี้ อีกทั้งตอนนี้ยังเป็นช่วงปลายฤดูกาลขายทรัฟเฟิล ย่อมไม่ต้องกังวลว่าจะขายไม่ออก

หลังตัดสินใจแล้ว เซี่ยรั่วเฟยก็นำกล่องเก็บของใบใหม่ออกมา ก่อนคัดทรัฟเฟิลดำและทรัฟเฟิลขาวก้อนใหญ่ที่สุดหลายก้อนออกมา แล้วเลือกทรัฟเฟิลขนาดปานกลางอีกสิบกว่าก้อน ใส่รวมกันลงในกล่องเก็บของใบนี้พร้อมเทข้าวสารลงไป

จากนั้นเซี่ยรั่วเฟยยังใช้เทปกาวปิดผนึกขอบกล่องเก็บของไว้อย่างแน่นหนา

ทรัฟเฟิลชุดนี้มีก้อนใหญ่ทั้งหมด 5 ก้อน แบ่งเป็นทรัฟเฟิลขาว 3 ก้อนและทรัฟเฟิลดำ 2 ก้อน ทรัฟเฟิลขาวล้วนหนักกว่า 3 จิน ส่วนทรัฟเฟิลดำก็หนักกว่า 2 จิน นอกจากนี้ยังมีทรัฟเฟิลทั่วไปอีก 13 ก้อน โดยแต่ละก้อนมีน้ำหนักราว 300-500 กรัม

ส่วนทรัฟเฟิลที่เหลือ เซี่ยรั่วเฟยก็ปิดผนึกเก็บไว้อย่างเรียบร้อย จากนั้นจึงพาสายฟ้าออกจากมิติภาพวิญญาณ

แม้จะวุ่นอยู่ในมิติภาพวิญญาณนานสิบกว่าชั่วโมง แต่ความจริงโลกภายนอกเพิ่งผ่านไปเพียงครู่เดียว เซี่ยรั่วเฟยเข้าไปล้างดินโคลนทั่วร่างในห้องน้ำก่อน หลังเปลี่ยนเป็นเสื้อผ้าสะอาดแล้วจึงเปิดประตู พาสายฟ้าลงไปชั้นล่างด้วยกัน

เซี่ยรั่วเฟยไม่เห็นหลินเฉี่ยว พอถามแม่ของหู่จื่อจึงรู้ว่าเด็กสาวคนนี้ยังคงหลับอุตุอยู่

ส่วนเย่หลิงอวิ๋นออกไปตรวจตราฟาร์มแล้ว—นี่เป็นสิ่งที่เขาต้องทำทุกวันเช่นกัน ในฐานะหัวหน้าฝ่ายรักษาความปลอดภัย เขาปฏิบัติหน้าที่อย่างเต็มความรับผิดชอบมาก

เซี่ยรั่วเฟยไม่มีอะไรทำ จึงพาสายฟ้าเดินเล่นไปทางฟาร์มเสียเลย

ยังไม่ทันเดินถึงโรงเรือนปลูกผัก เขาก็มองเห็นเฉาเถี่ยซู่เดินอย่างเร่งรีบออกมาจากโรงเรือนแต่ไกล…
